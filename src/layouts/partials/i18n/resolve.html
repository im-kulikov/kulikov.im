{{- /* Collapses {ru: .., en: ..} pairs to the current language (falls back to ru), recursively. */ -}} {{- $lang := site.Language.Lang -}} {{- $v := . -}} {{- if reflect.IsMap $v -}} {{- if isset $v "ru" -}} {{- $v = partial "i18n/resolve.html" (or (index $v $lang) (index $v "ru")) -}} {{- else -}} {{- $s := newScratch -}} {{- $s.Set "m" dict -}} {{- range $k, $x := $v -}}{{- $s.SetInMap "m" $k (partial "i18n/resolve.html" $x) -}}{{- end -}} {{- $v = $s.Get "m" -}} {{- end -}} {{- else if reflect.IsSlice $v -}} {{- $out := slice -}} {{- range $v -}}{{- $out = $out | append (slice (partial "i18n/resolve.html" .)) -}}{{- end -}} {{- $v = $out -}} {{- end -}} {{- return $v -}}
